- Node.js 18+
- Package Manager: npm / yarn / pnpm

## ۴. معماری فرانت‌اند

### ۴.۱. جریان داده

```
User Action → React State → useMutation() → Convex Cloud Function → Database
                                                                      ↓
User Screen ← React Re-render ← useQuery() ← Real-time Subscription ←
```

### ۴.۲. الگوی کامپوننت‌ها

```typescript
// الگوی استاندارد یک صفحه
export default function SomePage() {
  // ۱. خواندن داده از Convex (reactive)
  const data = useQuery(api.module.getData, { arg });

  // ۲. هوک‌های محلی
  const [filter, setFilter] = useState("");
  
  // ۳. هوک‌های سفارشی
  const { getVal, setVal } = useSettings("category");

  // ۴. mutation برای نوشتن
  const updateData = useMutation(api.module.updateData);

  // ۵. فیلتر و پردازش داده
  const filtered = data?.filter(item => 
    item.name.includes(filter)
  );

  // ۶. رندر
  return (
    <div>
      {/* فیلترها */}
      {/* جدول/کارت */}
      {/* مودال‌ها */}
    </div>
  );
}
```

## ۵. معماری بکند (Backend / Convex)

### ۵.۱. انواع توابع

| نوع | توضیح | مثال |
|-----|-------|------|
| `query` | خواندن داده (reactive) | `products.listActive` |
| `mutation` | نوشتن/بروزرسانی داده | `orders.updateStatus` |
| `action` | عملیات خارجی (HTTP, file, ...) | `smsNotifications.send` |
| `http` | endpoint‌های HTTP | `auth.addHttpRoutes` |

### ۵.۲. الگوی استاندارد یک فایل بکند

```typescript
// src/convex/products.ts
import { query, mutation } from "./_generated/server";
import { v } from "convex/values";

// ── لیست (خواندن) ──
export const listActive = query({
  args: {},                           // آرگومان‌ها
  handler: async (ctx) => {           // handler
    return await ctx.db
      .query("products")
      .withIndex("by_isActive", (q) => q.eq("isActive", true))
      .order("desc")
      .collect();                     // لیست نتایج
  },
});

// ── ایجاد (نوشتن) ──
export const create = mutation({
  args: {
    name: v.string(),
    price: v.number(),
    // ...
  },
  handler: async (ctx, args) => {
    // اعتبارسنجی
    if (args.price < 0) throw new Error("قیمت نمی‌تواند منفی باشد");
    
    // ثبت لاگ
    const userId = await getAuthUserId(ctx);
    await ctx.db.insert("auditLogs", {
      action: "create",
      entity: "product",
      entityId: "new",
      userId,
    });
    
    // ایجاد رکورد
    return await ctx.db.insert("products", {
      ...args,
      views: 0,
      createdAt: Date.now(),
    });
  },
});
```

### ۵.۳. realtime Subscriptions

```typescript
// در کلاینت - useQuery به صورت خودکار subscription ایجاد می‌کند
const products = useQuery(api.products.listActive);
// هر تغییر در دیتابیس → به‌روزرسانی خودکار UI

// در بکند - useQuery reactive است
// هیچ polling یا refetch دستی نیست
```

### ۵.۴. ایندکس‌گذاری

```typescript
// ایجاد ایندکس در schema.ts
products: defineTable({...})
  .index("by_category", ["categoryId"])
  .index("by_price", ["price"])

// استفاده از ایندکس در query
export const listByCategory = query({
  args: { categoryId: v.id("categories") },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("products")
      .withIndex("by_category", (q) => 
        q.eq("categoryId", args.categoryId)
      )
      .collect();
  },
});
```

## ۶. سیستم احراز هویت

### ۶.۱. جریان لاگین

```
کاربر → Auth.tsx → @convex-dev/auth → Convex Auth
                                    ↓
                              استفاده از یکی از:
                              • Email + Password
                              • Email OTP
                              • Guest (مهمان)
                              • Freebuff Federated JWT
```

### ۶.۲. فایل‌های کلیدی

| فایل | وظیفه |
|------|-------|
| `src/convex/auth.config.ts` | تنظیمات providers احراز هویت |
| `src/convex/auth.ts` | توابع لاگین/ثبت‌نام/خروج |
| `src/convex/auth/emailOtp.ts` | احراز هویت با کد OTP |
| `src/components/RequireAuth.tsx` | محافظ مسیر (redirect به /auth) |

newField: v.string()  // باعث خطا در رکوردهای قدیمی می‌شود

newField: v.string()  // و در handler مقدار پیش‌فرض تعیین کنید
